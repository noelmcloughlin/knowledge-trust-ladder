{{- /* The home page's five buttons, the same five in the same order on every site: the schema, the bundle, the architecture page, the glossary and the quick start. None is set anywhere. Each is looked for among the site's pages. The first one found is the filled button, and one that is not there is greyed, with the reason on hover. */ -}} {{- /* The quick start is a docs page of that name, or the first heading that says so: in the install page, then any docs page, then the README. */ -}} {{- $quick := "" -}} {{- range slice "/docs/quick-start" "/docs/quickstart" "/docs/getting-started" -}} {{- if not $quick }}{{ with site.GetPage . }}{{ $quick = .RelPermalink }}{{ end }}{{ end -}} {{- end -}} {{- $pages := slice -}} {{- with site.GetPage "/docs/install" }}{{ $pages = $pages | append . }}{{ end -}} {{- range where site.RegularPages "Section" "docs" }}{{ $pages = $pages | append . }}{{ end -}} {{- range $page := $pages | append site.Home -}} {{- range $id := slice "quick-start" "quickstart" "getting-started" "get-started" -}} {{- if and (not $quick) ($page.Fragments.Identifiers.Contains $id) }}{{ $quick = printf "%s#%s" $page.RelPermalink $id }}{{ end -}} {{- end -}} {{- end -}} {{- $link := dict -}} {{- range $key, $paths := dict "schema" (slice "/schema") "bundle" (slice "/knowledge") "architecture" (slice "/docs/architecture") "glossary" (slice "/knowledge/glossary" "/docs/glossary") -}} {{- range $paths }}{{ if not (index $link $key) }}{{ with site.GetPage . }}{{ $link = merge $link (dict $key .RelPermalink) }}{{ end }}{{ end }}{{ end -}} {{- end -}} {{- $filled := false -}}
